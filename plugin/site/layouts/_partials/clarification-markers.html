{{- /* Highlights `[NEEDS CLARIFICATION: ]` markers in rendered HTML, leaving code alone: each code element is swapped for a placeholder, the markers are wrapped, and the code is restored. Returns the HTML as a string. */ -}} {{- $html := . -}} {{- if strings.Contains $html "[NEEDS CLARIFICATION:" -}} {{- $code := findRE `(?s)]*>.*?` $html -}} {{- range $index, $element := $code -}} {{- $html = replace $html $element (printf "\x00code-%d\x00" $index) 1 -}} {{- end -}} {{- $marker := `Needs clarification $1` -}} {{- $html = replaceRE `\[NEEDS CLARIFICATION:\s*([^\]]+)\]` $marker $html -}} {{- range $index, $element := $code -}} {{- $html = replace $html (printf "\x00code-%d\x00" $index) $element 1 -}} {{- end -}} {{- end -}} {{- return $html -}}
