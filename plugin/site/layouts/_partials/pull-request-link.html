{{- /* A record's `pull-request`: a URL, or a number on `params.repository`. */ -}} {{- $value := string . -}} {{- $number := replaceRE `^.*/pull/(\d+)/?$` "$1" $value -}} {{- $href := "" -}} {{- if strings.HasPrefix $value "https://" -}} {{- $href = $value -}} {{- else if site.Params.repository -}} {{- $href = printf "%s/pull/%s" (strings.TrimSuffix "/" site.Params.repository) $value -}} {{- end -}} {{- $label := cond (findRE `^\d+$` $number) (printf "#%s" $number) $value -}} {{- with $href }}{{ $label }}{{ else }}{{ $label }}{{ end -}}
